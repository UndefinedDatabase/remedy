### `r31probe.py`

```python
"""F275 T002 — the DECISION F272 D7 descriptor probe, RECORDING disposition.

Loaded as a pytest plugin with `-p r31probe`. On import it installs a
class-level `property` over four fields: `id` and `name` on
`packages.core.models.Job`, and `job_id` and `job_title` on
`packages.orchestration.pingpong_job.JobPlan`.

Each property carries BOTH a getter and a setter. A getter-only property is
WRONG for a pydantic model with no `validate_assignment`: assignment writes
straight through to `__dict__`, and a setterless property raises
`AttributeError: can't set attribute` at every write site, reddening tests and
so measuring a SMALLER set than the truth.

A FIXED frame depth is equally wrong: pydantic dispatches assignment through
its own `__setattr__`, so the caller two frames out is `pydantic/main.py`.
Recording therefore attributes the access to the INNERMOST STACK FRAME WHOSE
FILENAME LIES UNDER THE WORKTREE ROOT, walking outward from the caller, and
falls back to the innermost frame of all when no frame qualifies.

Record key: (owner, field, mode, direct, repo-relative path, line, function).
Value: the access count. Written as JSON at `pytest_sessionfinish`.
"""
import json
import os
import sys

ROOT = os.path.dirname(os.path.abspath(__file__))
OUT = os.environ.get("R31_PROBE_OUT", os.path.join(ROOT, "r31_probe_sites.json"))

# key -> count
RECORDS: dict = {}
# filename -> (qualifies, repo-relative path)
_FILE_CACHE: dict = {}

_PREFIX = ROOT + os.sep


def _classify(filename):
    hit = _FILE_CACHE.get(filename)
    if hit is None:
        try:
            ab = os.path.abspath(filename)
        except Exception:
            ab = filename
        if ab.startswith(_PREFIX):
            hit = (True, ab[len(_PREFIX):])
        else:
            hit = (False, ab)
        _FILE_CACHE[filename] = hit
    return hit


def _record(owner, field, mode):
    frame = sys._getframe(2)          # 0 = _record, 1 = getter/setter, 2 = caller
    if frame is None:                 # pragma: no cover - defensive
        return
    innermost = frame
    direct = None
    chosen = None
    f = frame
    while f is not None:
        qual, rel = _classify(f.f_code.co_filename)
        if direct is None:
            direct = qual
        if qual:
            chosen = (rel, f.f_lineno, f.f_code.co_name)
            break
        f = f.f_back
    if chosen is None:
        _, rel = _classify(innermost.f_code.co_filename)
        chosen = (rel, innermost.f_lineno, innermost.f_code.co_name)
    key = (owner, field, mode, bool(direct)) + chosen
    RECORDS[key] = RECORDS.get(key, 0) + 1


def _make(owner, field, pydantic):
    def getter(self, _o=owner, _f=field):
        _record(_o, _f, "read")
        return self.__dict__[_f]

    def setter(self, value, _o=owner, _f=field, _p=pydantic):
        _record(_o, _f, "write")
        self.__dict__[_f] = value
        if _p:
            # what pydantic's own no-`validate_assignment` path does
            self.__pydantic_fields_set__.add(_f)

    return property(getter, setter)


def install():
    from packages.core.models import Job
    from packages.orchestration.pingpong_job import JobPlan

    for mod in (sys.modules["packages.core.models"],
                sys.modules["packages.orchestration.pingpong_job"]):
        qual, _ = _classify(mod.__file__)
        if not qual:
            raise RuntimeError(
                f"r31probe: {mod.__name__} resolves OUTSIDE the worktree "
                f"root {ROOT} -> {mod.__file__}"
            )

    for field in ("id", "name"):
        setattr(Job, field, _make("Job", field, True))
    for field in ("job_id", "job_title"):
        setattr(JobPlan, field, _make("JobPlan", field, False))


install()


def pytest_report_header(config):
    import packages.core.models as m
    import packages.orchestration.pingpong_job as p
    return [
        f"r31probe: root={ROOT}",
        f"r31probe: models={m.__file__}",
        f"r31probe: pingpong_job={p.__file__}",
        f"r31probe: out={OUT}",
    ]


def pytest_sessionfinish(session, exitstatus):
    rows = [{"owner": k[0], "field": k[1], "mode": k[2], "direct": k[3],
             "path": k[4], "line": k[5], "func": k[6], "count": v}
            for k, v in RECORDS.items()]
    rows.sort(key=lambda r: (r["owner"], r["field"], r["mode"], r["path"],
                             r["line"], r["func"], r["direct"]))
    with open(OUT, "w") as fh:
        json.dump({"root": ROOT, "exitstatus": int(exitstatus), "rows": rows},
                  fh, indent=1)
```

### `r31_static.py`

```python
"""F275 T002 — the static half of the site set: an `ast` reading over every
tracked `.py` from `git ls-files '*.py'`, enumerated from git and never from a
shell glob, per DECISION F272 D2. Per scope it binds each local NAME to a
verdict on whether it holds a `Job` — `job` from a parameter or `AnnAssign`
annotation or an assignment from a `Job(...)` call, `other` from an annotation
naming something else, `unknown` otherwise — then records every `Attribute`
node whose `attr` is `id` or `name` over a bare `Name`, with that verdict, the
path, the line and the load/store context. It also counts the rest of the same
atomic commit: `Job(...)` constructions, `Job` imports, `Job` annotations.
"""
import ast
import json
import os
import subprocess
import sys

ROOT = os.path.dirname(os.path.abspath(__file__))
OUT = os.environ.get("R31_STATIC_OUT", os.path.join(ROOT, "r31_static_sites.json"))
FIELDS = ("id", "name")
SCOPES = (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef, ast.Lambda)
BODIES = ("body", "orelse", "finalbody", "handlers")


def tracked_py():
    out = subprocess.run(["git", "-C", ROOT, "ls-files", "*.py"],
                         capture_output=True, text=True, check=True).stdout
    return [ln for ln in out.split("\n") if ln]


def annotation_names(node, depth=0):
    """Every identifier an annotation expression mentions, strings included."""
    names = set()
    if node is None or depth > 3:
        return names
    for n in ast.walk(node):
        if isinstance(n, ast.Name):
            names.add(n.id)
        elif isinstance(n, ast.Attribute):
            names.add(n.attr)
        elif isinstance(n, ast.Constant) and isinstance(n.value, str):
            try:
                names |= annotation_names(ast.parse(n.value, mode="eval").body,
                                          depth + 1)
            except SyntaxError:
                pass
    return names


def call_is_job(node):
    if not isinstance(node, ast.Call):
        return False
    f = node.func
    return (isinstance(f, ast.Name) and f.id == "Job") or \
           (isinstance(f, ast.Attribute) and f.attr == "Job")


def params(node):
    a = node.args
    out = list(a.posonlyargs) + list(a.args) + list(a.kwonlyargs)
    return out + [x for x in (a.vararg, a.kwarg) if x is not None]


def scope_statements(scope):
    """Statements lexically owned by this scope, nested scopes excluded."""
    out, stack = [], []
    for f in BODIES:
        v = getattr(scope, f, None)
        if isinstance(v, list):
            stack.extend(v)
    while stack:
        s = stack.pop()
        if isinstance(s, SCOPES):
            continue
        out.append(s)
        for f in BODIES:
            v = getattr(s, f, None)
            if isinstance(v, list):
                stack.extend(v)
    return out


def bind(scope):
    """NAME -> verdict, for one scope."""
    verdicts = {}

    def mark(name, v):
        if v == "job" or verdicts.get(name) != "job":
            verdicts[name] = "job" if v == "job" else v

    if isinstance(scope, (ast.FunctionDef, ast.AsyncFunctionDef, ast.Lambda)):
        for arg in params(scope):
            names = annotation_names(getattr(arg, "annotation", None))
            if names:
                mark(arg.arg, "job" if "Job" in names else "other")
    for s in scope_statements(scope):
        if isinstance(s, ast.AnnAssign) and isinstance(s.target, ast.Name):
            names = annotation_names(s.annotation)
            if names:
                mark(s.target.id, "job" if "Job" in names else "other")
            if s.value is not None and call_is_job(s.value):
                mark(s.target.id, "job")
        elif isinstance(s, ast.Assign) and call_is_job(s.value):
            for t in s.targets:
                if isinstance(t, ast.Name):
                    mark(t.id, "job")
    return verdicts


def walk_scope(scope, path, sites):
    """Record FIELD sites owned by this scope, then recurse into nested ones."""
    verdicts = bind(scope)
    stack, nested = list(ast.iter_child_nodes(scope)), []
    while stack:
        n = stack.pop()
        if isinstance(n, SCOPES):
            nested.append(n)
            continue
        if isinstance(n, ast.Attribute) and n.attr in FIELDS and \
                isinstance(n.value, ast.Name):
            sites.append({
                "path": path, "line": n.lineno, "field": n.attr,
                "recv": n.value.id,
                "verdict": verdicts.get(n.value.id, "unknown"),
                "ctx": "store" if isinstance(n.ctx, ast.Store) else "load",
            })
        stack.extend(ast.iter_child_nodes(n))
    for n in nested:
        walk_scope(n, path, sites)


def tally(tree, path, counts):
    """Constructions, imports and annotations of `Job` in one module."""
    def add(kind, line):
        counts[kind].append({"path": path, "line": line})

    for n in ast.walk(tree):
        if call_is_job(n):
            add("constructions", n.lineno)
        elif isinstance(n, (ast.Import, ast.ImportFrom)):
            for a in n.names:
                if a.name.split(".")[-1] == "Job":
                    add("imports", n.lineno)
        elif isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef)):
            for arg in params(n):
                if "Job" in annotation_names(arg.annotation):
                    add("annotations", arg.lineno)
            if "Job" in annotation_names(n.returns):
                add("annotations", n.lineno)
        elif isinstance(n, ast.AnnAssign) and "Job" in annotation_names(n.annotation):
            add("annotations", n.lineno)


def main():
    paths, sites, failed, parsed = tracked_py(), [], [], 0
    counts = {"constructions": [], "imports": [], "annotations": []}
    for p in paths:
        try:
            tree = ast.parse(open(os.path.join(ROOT, p), "rb").read(), filename=p)
        except (SyntaxError, OSError, ValueError) as exc:
            failed.append([p, str(exc)])
            continue
        parsed += 1
        walk_scope(tree, p, sites)
        tally(tree, p, counts)

    with open(OUT, "w") as fh:
        json.dump({"tracked_py": len(paths), "parsed": parsed, "failed": failed,
                   "sites": sites, "counts": counts}, fh, indent=1)
    print("tracked .py from git ls-files:", len(paths), "parsed:", parsed,
          "unparsable:", len(failed))
    print("attribute sites recorded:", len(sites))
    for v in ("job", "other", "unknown"):
        print(f"  verdict {v:<8} {sum(1 for s in sites if s['verdict'] == v)}")
    for k, items in counts.items():
        print(f"  {k:<14} {len(items)}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
