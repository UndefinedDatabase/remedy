"""F275 R47 — the FIRST slice of the id-shape migration: `TaskExecutionContext`.

import ast
import collections
import pathlib
import subprocess

FIELDS = {("TaskExecutionContext", "job_id"), ("TaskExecutionContext", "task_id")}
MODELS = {m for m, _ in FIELDS}


def tracked():
    out = subprocess.run(["git", "ls-files", "*.py"], capture_output=True,
                         text=True).stdout
    return [p for p in out.split() if p]


def apply_edits(src, edits):
    lines = src.split("\n")
    by_line = collections.defaultdict(list)
    for lineno, a, b, new in edits:
        by_line[lineno].append((a, b, new))
    for lineno, items in by_line.items():
        raw = lines[lineno - 1].encode("utf-8")
        for a, b, new in sorted(set(items), key=lambda x: -x[0]):
            raw = raw[:a] + new.encode("utf-8") + raw[b:]
        lines[lineno - 1] = raw.decode("utf-8")
    return "\n".join(lines)


counts = collections.Counter()
written = 0
for rel in tracked():
    p = pathlib.Path(rel)
    src = p.read_text()
    try:
        tree = ast.parse(src, filename=rel)
    except SyntaxError:
        continue
    edits = []
    for n in ast.walk(tree):
        if isinstance(n, ast.ClassDef) and n.name in MODELS:
            for stmt in n.body:
                if (isinstance(stmt, ast.AnnAssign)
                        and isinstance(stmt.target, ast.Name)
                        and (n.name, stmt.target.id) in FIELDS):
                    ann = stmt.annotation
                    edits.append((ann.lineno, ann.col_offset, ann.end_col_offset, "str"))
                    counts[f"decl {n.name}.{stmt.target.id}"] += 1
        if isinstance(n, ast.Call):
            f = n.func
            called = f.id if isinstance(f, ast.Name) else (
                f.attr if isinstance(f, ast.Attribute) else None)
            if called not in MODELS:
                continue
            for k in n.keywords:
                if k.arg is None or (called, k.arg) not in FIELDS:
                    continue
                v = k.value
                if isinstance(v, ast.Constant):
                    continue
                if (isinstance(v, ast.Call) and isinstance(v.func, ast.Name)
                        and v.func.id == "str"):
                    continue
                if v.lineno != v.end_lineno:
                    counts["writer SKIPPED multiline"] += 1
                    continue
                edits.append((v.lineno, v.col_offset, v.col_offset, "str("))
                edits.append((v.end_lineno, v.end_col_offset, v.end_col_offset, ")"))
                counts[f"writer {called}.{k.arg}"] += 1
    if not edits:
        continue
    out = apply_edits(src, edits)
    try:
        ast.parse(out, filename=rel)
    except SyntaxError as exc:
        print(f"  BROKE {rel}: {exc}")
        continue
    p.write_text(out)
    written += 1

print(f"files rewritten: {written}")
for k, v in sorted(counts.items()):
    print(f"  {k:38s} {v}")
print(f"total edits: {sum(counts.values())}")
